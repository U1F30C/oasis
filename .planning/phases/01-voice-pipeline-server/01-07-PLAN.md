**`tests/ws-server.test.ts`** (REQ-02, REQ-07 — WebSocket server accepts connections and routes messages):

This test creates an in-process WebSocket server and connects a ws client to it. The pipeline is mocked so no model downloads occur. Verifies: connect → "ready" received → send PCM binary → receive WAV binary response.

```typescript
import { describe, it, expect, vi, beforeAll, afterAll } from "vitest";
import { createServer } from "http";
import { WebSocketServer, WebSocket } from "ws";

// Mock all pipeline dependencies before importing the handler
vi.mock("../src/server/pipeline.js", () => ({
  getSttPipeline: vi.fn().mockReturnValue(
    vi.fn().mockResolvedValue({ text: "mocked stt output" }),
  ),
  getTtsPipeline: vi.fn().mockReturnValue(
    vi.fn().mockResolvedValue({
      sampling_rate: 16000,
      audio: new Float32Array(16000),
    }),
  ),
  SPEAKER_EMBEDDINGS: "mock-embeddings-url",
}));

vi.mock("../src/server/llm.js", () => ({
  gemma3n: "mock-model",
  generateText: vi.fn().mockResolvedValue({ text: "mocked llm response" }),
}));

vi.mock("../src/server/tts.js", () => {
  const { WaveFile } = require("wavefile");
  return {
    generateWav: vi.fn().mockImplementation(
      (ttsOut: { sampling_rate: number; audio: Float32Array }) => {
        const wav = new WaveFile();
        wav.fromScratch(1, ttsOut.sampling_rate, "32f", ttsOut.audio);
        wav.toBitDepth("16");
        return wav;
      },
    ),
  };
});

const { handleConnection } = await import("../src/server/routes/ws.js");

describe("REQ-02 / REQ-07: WebSocket server connection and pipeline dispatch", () => {
  let server: ReturnType<typeof createServer>;
  let wss: WebSocketServer;
  let port: number;

  beforeAll(async () => {
    server = createServer();
    wss = new WebSocketServer({ noServer: true });

    server.on("upgrade", (req, socket, head) => {
      wss.handleUpgrade(req, socket, head, (ws) => wss.emit("connection", ws, req));
    });

    wss.on("connection", handleConnection);

    await new Promise<void>((resolve) => {
      server.listen(0, () => {
        port = (server.address() as { port: number }).port;
        resolve();
      });
    });
  });

  afterAll(async () => {
    await new Promise<void>((resolve) => wss.close(() => resolve()));
    await new Promise<void>((resolve) => server.close(() => resolve()));
  });

  it("sends { type: 'ready' } on connect", async () => {
    const ws = new WebSocket(`ws://localhost:${port}`);

    const readyMsg = await new Promise<string>((resolve, reject) => {
      ws.once("message", (data) => resolve(data.toString()));
      ws.once("error", reject);
    });

    const parsed = JSON.parse(readyMsg);
    expect(parsed.type).toBe("ready");

    ws.close();
  });

  it("accepts Float32Array PCM binary and responds with WAV binary", async () => {
    const ws = new WebSocket(`ws://localhost:${port}`);
    ws.binaryType = "arraybuffer";

    // Wait for "ready"
    await new Promise<void>((resolve, reject) => {
      ws.once("message", () => resolve());
      ws.once("error", reject);
    });

    // Send Float32Array PCM (not a WAV header — raw PCM path)
    const pcm = new Float32Array(16000); // 1s of silence
    ws.send(pcm.buffer);

    // Collect responses until we get a binary frame (WAV)
    const wavBuffer = await new Promise<ArrayBuffer>((resolve, reject) => {
      const timeout = setTimeout(() => reject(new Error("Timeout waiting for WAV response")), 5000);
      ws.on("message", (data: Buffer | ArrayBuffer) => {
        const buf = data instanceof Buffer ? data.buffer : data;
        if (buf.byteLength > 44) {
          clearTimeout(timeout);
          resolve(buf);
        }
      });
      ws.once("error", reject);
    });

    // Check RIFF WAV header
    const view = new Uint8Array(wavBuffer);
    expect(view[0]).toBe(0x52); // R
    expect(view[1]).toBe(0x49); // I
    expect(view[2]).toBe(0x46); // F
    expect(view[3]).toBe(0x46); // F

    ws.close();
  });

  it("responds to ping with pong", async () => {
    const ws = new WebSocket(`ws://localhost:${port}`);

    // Wait for ready
    await new Promise<void>((resolve, reject) => {
      ws.once("message", () => resolve());
      ws.once("error", reject);
    });

    // Send ping
    ws.send(JSON.stringify({ type: "ping" }));

    const pongMsg = await new Promise<string>((resolve, reject) => {
      ws.once("message", (data) => resolve(data.toString()));
      ws.once("error", reject);
    });

    expect(JSON.parse(pongMsg).type).toBe("pong");

    ws.close();
  });
});
```

**`tests/state-machine.test.ts`** (REQ-05 — useVoiceWS hook state transitions):

This test exercises the state machine logic in useVoiceWS. It uses renderHook from @testing-library/react (requires `npm install --save-dev @testing-library/react`). The test environment is "jsdom" for this file — add a vitest environment docblock.

```typescript
// @vitest-environment jsdom
import { describe, it, expect, vi, beforeEach } from "vitest";
import { renderHook, act } from "@testing-library/react";

// Mock WebSocket
class MockWebSocket {
  binaryType = "arraybuffer";
  readyState = 1; // OPEN
  onopen: (() => void) | null = null;
  onmessage: ((e: MessageEvent) => void) | null = null;
  onerror: ((e: Event) => void) | null = null;
  onclose: (() => void) | null = null;
  sentMessages: (string | ArrayBuffer)[] = [];

  send(data: string | ArrayBuffer) {
    this.sentMessages.push(data);
  }

  close() {
    if (this.onclose) this.onclose();
  }

  // Test helper: simulate receiving a text message
  simulateTextMessage(data: string) {
    if (this.onmessage) {
      this.onmessage(new MessageEvent("message", { data }));
    }
  }

  // Test helper: simulate receiving a binary message (WAV)
  simulateBinaryMessage(data: ArrayBuffer) {
    if (this.onmessage) {
      this.onmessage(new MessageEvent("message", { data }));
    }
  }
}

let mockWsInstance: MockWebSocket;

vi.stubGlobal(
  "WebSocket",
  vi.fn().mockImplementation((url: string) => {
    mockWsInstance = new MockWebSocket();
    // Trigger onopen asynchronously (simulates real WS behavior)
    setTimeout(() => mockWsInstance.onopen?.(), 0);
    return mockWsInstance;
  }),
);

// Mock getUserMedia and MediaRecorder for recording tests
vi.stubGlobal("navigator", {
  mediaDevices: {
    getUserMedia: vi.fn().mockResolvedValue({
      getTracks: () => [{ stop: vi.fn() }],
    }),
  },
});

vi.stubGlobal(
  "MediaRecorder",
  vi.fn().mockImplementation(() => ({
    start: vi.fn(),
    stop: vi.fn(function (this: { onstop?: () => void }) {
      this.onstop?.();
    }),
    state: "inactive",
    mimeType: "audio/webm",
    ondataavailable: null,
    onstop: null,
  })),
);

// Import after globals are mocked
const { useVoiceWS } = await import("../src/client/src/hooks/useVoiceWS.js");

describe("REQ-05: useVoiceWS state machine transitions", () => {
  beforeEach(() => {
    vi.clearAllMocks();
  });

  it("initial state is 'idle'", () => {
    const { result } = renderHook(() => useVoiceWS());
    expect(result.current.state).toBe("idle");
  });

  it("connect() opens WebSocket and state remains 'idle'", async () => {
    const { result } = renderHook(() => useVoiceWS());

    act(() => {
      result.current.connect("ws://localhost:3001/ws");
    });

    // Let onopen fire
    await act(async () => {
      await new Promise((r) => setTimeout(r, 10));
    });

    expect(result.current.state).toBe("idle");
    expect(result.current.error).toBeNull();
  });

  it("server 'processing' message transitions state to 'processing'", async () => {
    const { result } = renderHook(() => useVoiceWS());

    act(() => result.current.connect("ws://localhost:3001/ws"));
    await act(async () => { await new Promise((r) => setTimeout(r, 10)); });

    await act(async () => {
      mockWsInstance.simulateTextMessage(JSON.stringify({ type: "processing" }));
    });

    expect(result.current.state).toBe("processing");
  });

  it("binary WAV message transitions state to 'playing' then 'idle'", async () => {
    // Mock playWavBuffer to resolve immediately
    vi.doMock("../src/client/src/audio.js", () => ({
      captureAndSendAudio: vi.fn(),
      playWavBuffer: vi.fn().mockResolvedValue(undefined),
    }));

    const { result } = renderHook(() => useVoiceWS());
    act(() => result.current.connect("ws://localhost:3001/ws"));
    await act(async () => { await new Promise((r) => setTimeout(r, 10)); });

    const fakeWav = new ArrayBuffer(100);

    await act(async () => {
      mockWsInstance.simulateBinaryMessage(fakeWav);
      await new Promise((r) => setTimeout(r, 50));
    });

    // After playWavBuffer resolves, state should return to idle
    expect(result.current.state).toBe("idle");
  });

  it("server 'error' message sets error state and returns to 'idle'", async () => {
    const { result } = renderHook(() => useVoiceWS());
    act(() => result.current.connect("ws://localhost:3001/ws"));
    await act(async () => { await new Promise((r) => setTimeout(r, 10)); });

    await act(async () => {
      mockWsInstance.simulateTextMessage(JSON.stringify({ type: "error", message: "busy" }));
    });

    expect(result.current.state).toBe("idle");
    expect(result.current.error).toBe("busy");
  });
});
```
