```typescript
import { describe, it, expect } from "vitest";
import { readAudio } from "../src/server/stt.js";
import { generateWav } from "../src/server/tts.js";
import { gemma3n, qwen3wen3_8b, gemma3_270m } from "../src/server/llm.js";
import {
  getSttPipeline,
  getTtsPipeline,
  SPEAKER_EMBEDDINGS,
} from "../src/server/pipeline.js";

describe("REQ-10: Server lib exports (stt, tts, llm)", () => {
  it("readAudio is an exported async function", () => {
    expect(typeof readAudio).toBe("function");
    // It's async — returns a Promise
    expect(readAudio.constructor.name === "AsyncFunction" || readAudio.toString().includes("async")).toBe(true);
  });

  it("generateWav is an exported function", () => {
    expect(typeof generateWav).toBe("function");
  });

  it("generateWav produces a WaveFile with 16-bit samples", () => {
    // Create minimal Float32Array audio
    const sampleRate = 16000;
    const audio = new Float32Array(sampleRate); // 1 second of silence
    const wav = generateWav({ sampling_rate: sampleRate, audio });

    // WaveFile instance should have toBuffer method
    expect(typeof wav.toBuffer).toBe("function");

    // Should have been converted to 16-bit (bitDepth "16")
    const buffer = wav.toBuffer() as Buffer;
    expect(buffer.length).toBeGreaterThan(44); // WAV header is 44 bytes
  });

  it("llm.ts exports three Ollama models (not undefined)", () => {
    expect(gemma3n).toBeDefined();
    expect(qwen3wen3_8b).toBeDefined();
    expect(gemma3_270m).toBeDefined();
  });
});

describe("REQ-09: Pipeline guards throw before initPipelines()", () => {
  it("getSttPipeline throws before initialization", () => {
    expect(() => getSttPipeline()).toThrow("STT pipeline not initialized");
  });

  it("getTtsPipeline throws before initialization", () => {
    expect(() => getTtsPipeline()).toThrow("TTS pipeline not initialized");
  });

  it("SPEAKER_EMBEDDINGS is a string URL pointing to huggingface.co", () => {
    expect(typeof SPEAKER_EMBEDDINGS).toBe("string");
    expect(SPEAKER_EMBEDDINGS).toContain("huggingface.co");
    expect(SPEAKER_EMBEDDINGS).toContain("speaker_embeddings");
  });
});
```

**`tests/protocol.test.ts`:**
```typescript
import { describe, it, expect } from "vitest";
import type { ControlMessage, PipelineState } from "../src/shared/protocol.js";

// These are TypeScript type-level tests. We verify the discriminated union
// shapes are correct by constructing values and asserting their properties.

describe("REQ-03: Shared WebSocket protocol types", () => {
  it("ControlMessage: ready shape is valid", () => {
    const msg: ControlMessage = { type: "ready" };
    expect(msg.type).toBe("ready");
  });

  it("ControlMessage: processing shape is valid", () => {
    const msg: ControlMessage = { type: "processing" };
    expect(msg.type).toBe("processing");
  });

  it("ControlMessage: error shape carries message field", () => {
    const msg: ControlMessage = { type: "error", message: "something failed" };
    expect(msg.type).toBe("error");
    expect((msg as Extract<ControlMessage, { type: "error" }>).message).toBe("something failed");
  });

  it("ControlMessage: ping and pong shapes are valid", () => {
    const ping: ControlMessage = { type: "ping" };
    const pong: ControlMessage = { type: "pong" };
    expect(ping.type).toBe("ping");
    expect(pong.type).toBe("pong");
  });

  it("PipelineState covers all four states", () => {
    const states: PipelineState[] = ["idle", "recording", "processing", "playing"];
    expect(states).toHaveLength(4);
    states.forEach((s) => expect(typeof s).toBe("string"));
  });
});
```
