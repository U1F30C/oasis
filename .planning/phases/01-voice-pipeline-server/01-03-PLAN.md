```typescript
type Pipeline = Awaited<ReturnType<typeof pipeline>>;

interface PipelineInstances {

}

const instances: PipelineInstances = { stt: null, tts: null };

  console.log("Loading STT pipeline (Whisper whisper-tiny.en)...");
  instances.stt = await pipeline(
    "automatic-speech-recognition",
    "Xenova/whisper-tiny.en",
    { dtype: { encoder_model: "fp32", decoder_model_merged: "q4" } },
  );

  console.log("Loading TTS pipeline (speecht5_tts)...");
  instances.tts = await pipeline("text-to-speech", "Xenova/speecht5_tts");

  console.log("Pipelines ready.");
}

export function getSttPipeline(): Pipeline {
  if (!instances.stt) {
    throw new Error("STT pipeline not initialized — call initPipelines() first");
  }
  return instances.stt;
}

export function getTtsPipeline(): Pipeline {
  if (!instances.tts) {
    throw new Error("TTS pipeline not initialized — call initPipelines() first");
  }
  return instances.tts;
}
```

Create `src/server/routes/ws.ts` — the WebSocket connection handler that receives binary audio, runs STT→LLM→TTS, and sends back 16-bit WAV bytes.

The handler must implement ALL of the following behaviors (derived from RESEARCH.md Patterns 3-5 and index.ts processRecording):

**RIFF magic-byte detection (IoT WAV path fix):**
The binary frame may be either:
- Raw Float32Array PCM bytes (browser path: OfflineAudioContext output)
- A complete WAV file (IoT path: `arecord -t wav ...` → `ws.send(fs.readFileSync(file))`)

Detect which format by inspecting the first 4 bytes for the RIFF header magic:
```typescript
// RIFF detection: bytes 0-3 === 0x52 0x49 0x46 0x46 ("RIFF")
function isWavBuffer(buf: Buffer): boolean {
  return buf.length >= 4 &&
    buf[0] === 0x52 && buf[1] === 0x49 && buf[2] === 0x46 && buf[3] === 0x46;
}
```

When RIFF is detected:
- Write the buffer to a temp file using `os.tmpdir()` + a unique filename
- Call `readAudio(tempPath, 16000)` to get a Float32Array
- Delete the temp file after reading

When RIFF is NOT detected:
- Treat the buffer as raw Float32Array PCM bytes (browser path):
  `const audioData = new Float32Array(data.buffer, data.byteOffset, data.byteLength / 4);`

**Busy guard (RESEARCH.md Pitfall 1):** Use a module-level `let isBusy = false` flag. If a new binary message arrives while isBusy, immediately send `{ type: "error", message: "busy" }` and return.

**readyState guard (RESEARCH.md Pitfall 2):** Check `ws.readyState === WebSocket.OPEN` before every `ws.send()` call.

**STT result extraction (index.ts lines 196-198):** Copy verbatim including the Array.isArray ternary.

**Short input guard (index.ts lines 203-207):** If `!userText || userText.trim().length < 2`, send error and return early.

**LLM prompt (index.ts line 221):** Copy verbatim — `"You are a helpful voice assistant. Keep responses concise and conversational and very short, one or two sentences. User said: "${userText}"`

**WAV encoding:** After TTS: `const wav = generateWav(ttsOut)` → `wav.toBuffer()` → send as binary.

```typescript
import os from "os";
import path from "path";
import fs from "fs";
import { WebSocket } from "ws";
import { generateText } from "ai";
import { getSttPipeline, getTtsPipeline, SPEAKER_EMBEDDINGS } from "../pipeline.js";
import { generateWav } from "../tts.js";
import { readAudio } from "../stt.js";
import { gemma3n } from "../llm.js";
import type { ControlMessage } from "../../shared/protocol.js";

let isBusy = false;

function send(ws: WebSocket, msg: ControlMessage): void {
  if (ws.readyState === WebSocket.OPEN) {
    ws.send(JSON.stringify(msg));
  }
}

// Detect WAV file by RIFF magic bytes (0x52 0x49 0x46 0x46)
function isWavBuffer(buf: Buffer): boolean {
  return (
    buf.length >= 4 &&
    buf[0] === 0x52 &&
    buf[1] === 0x49 &&
    buf[2] === 0x46 &&
    buf[3] === 0x46
  );
}

async function bufferToFloat32(data: Buffer): Promise<Float32Array> {
  if (isWavBuffer(data)) {
    // IoT path: WAV file bytes — write to temp file and decode via readAudio()
    const tmpPath = path.join(os.tmpdir(), `oasis-${Date.now()}.wav`);
    fs.writeFileSync(tmpPath, data);
    try {
      return await readAudio(tmpPath, 16000);
    } finally {
      fs.unlinkSync(tmpPath);
    }
  }
  // Browser path: raw Float32Array PCM bytes (little-endian, 16kHz, mono)
  return new Float32Array(data.buffer, data.byteOffset, data.byteLength / 4);
}

async function runPipeline(data: Buffer): Promise<Buffer> {
  // 1. Decode audio (WAV or raw PCM)
  const audioData = await bufferToFloat32(data);

  // 2. STT
  const sttPipe = getSttPipeline();
  const sttResult = await sttPipe(audioData, { chunk_length_s: 10 });
  const userText = Array.isArray(sttResult)
    ? sttResult[0]?.text || ""
    : (sttResult as { text: string }).text;

  console.log(`[STT] "${userText}"`);

  if (!userText || userText.trim().length < 2) {
    throw new Error("No clear input detected");
  }

  // 3. LLM
  const { text: llmResponse } = await generateText({
    model: gemma3n,
    prompt: `You are a helpful voice assistant. Keep responses concise and conversational and very short, one or two sentences. User said: "${userText}"`,
  });

  console.log(`[LLM] "${llmResponse}"`);

  // 4. TTS
  const ttsPipe = getTtsPipeline();
  const ttsOut = await ttsPipe(llmResponse, { speaker_embeddings: SPEAKER_EMBEDDINGS });

  // 5. Encode as 16-bit WAV (generateWav already calls toBitDepth("16"))
  const wav = generateWav(ttsOut as { sampling_rate: number; audio: Float32Array });
  return wav.toBuffer() as Buffer;
}

export function handleConnection(ws: WebSocket): void {
  // Notify client pipelines are ready
  send(ws, { type: "ready" });

  ws.on("message", async (data: Buffer, isBinaryFrame: boolean) => {
    if (!isBinaryFrame) {
      // JSON control message
      try {
        const msg = JSON.parse(data.toString()) as ControlMessage;
        if (msg.type === "ping") send(ws, { type: "pong" });
      } catch {
        // Ignore malformed JSON
      }
      return;
    }

    // Binary = audio payload
    if (isBusy) {
      send(ws, { type: "error", message: "busy" });
      return;
    }

    isBusy = true;
    send(ws, { type: "processing" });

    try {
      const wavBuffer = await runPipeline(data);
      if (ws.readyState === WebSocket.OPEN) {
        ws.send(wavBuffer);
      }
    } catch (error) {
      console.error("[Pipeline error]", error);
      send(ws, { type: "error", message: String(error) });
    } finally {
      isBusy = false;
    }
  });

  ws.on("close", () => {
    console.log("[WS] Client disconnected");
    isBusy = false;
  });

  ws.on("error", (err) => {
    console.error("[WS] Error:", err);
    isBusy = false;
  });
}
```

Create `src/server/routes/api.ts` — Express Router exposing each pipeline stage as an independent HTTP endpoint. These let IoT devices and CLI tools call individual stages without a WebSocket connection (REQ-09).

**Routes to implement:**

**`POST /api/stt`** — accepts raw binary audio body (WAV file bytes OR Float32 PCM) and returns `{ text: string }`:
- Use `express.raw({ type: "application/octet-stream", limit: "10mb" })` middleware on this route
- Also accept `multipart/form-data` with `audio` field via a manual buffer read (use `req.on("data")` accumulation if Content-Type is multipart) — but for v1, accept ONLY `application/octet-stream` binary body; IoT clients send raw WAV bytes with this content type
- Apply the same RIFF detection logic as ws.ts: if buffer starts with RIFF magic, write to temp file and call `readAudio(tmpPath, 16000)`; otherwise treat as Float32Array PCM
- Run STT via `getSttPipeline()`
- Return `{ text: string }`

**`POST /api/llm`** — accepts `{ prompt: string }` JSON body and returns `{ response: string }`:
- Use `express.json()` (already mounted globally in index.ts, but safe to apply per-route too)
- Call `generateText({ model: gemma3n, prompt: req.body.prompt })`
- Return `{ response: text }`

**`POST /api/tts`** — accepts `{ text: string }` JSON body and returns 16-bit WAV binary:
- Call `getTtsPipeline()(text, { speaker_embeddings: SPEAKER_EMBEDDINGS })` 
- Call `generateWav(ttsOut)` → `wav.toBuffer()`
- Set `Content-Type: audio/wav` and send the buffer as binary response

```typescript
import os from "os";
import path from "path";
import fs from "fs";
import { Router, Request, Response } from "express";
import { generateText } from "ai";
import { getSttPipeline, getTtsPipeline, SPEAKER_EMBEDDINGS } from "../pipeline.js";
import { generateWav } from "../tts.js";
import { readAudio } from "../stt.js";
import { gemma3n } from "../llm.js";

export const apiRouter = Router();

// Detect WAV file by RIFF magic bytes (shared logic with ws.ts)
function isWavBuffer(buf: Buffer): boolean {
  return (
    buf.length >= 4 &&
    buf[0] === 0x52 &&
    buf[1] === 0x49 &&
    buf[2] === 0x46 &&
    buf[3] === 0x46
  );
}

async function bufferToFloat32(data: Buffer): Promise<Float32Array> {
  if (isWavBuffer(data)) {
    const tmpPath = path.join(os.tmpdir(), `oasis-stt-${Date.now()}.wav`);
    fs.writeFileSync(tmpPath, data);
    try {
      return await readAudio(tmpPath, 16000);
    } finally {
      fs.unlinkSync(tmpPath);
    }
  }
  return new Float32Array(data.buffer, data.byteOffset, data.byteLength / 4);
}

/**
 * POST /api/stt
 * Body: raw binary audio (WAV file bytes or Float32Array PCM)
 * Content-Type: application/octet-stream
 * Response: { text: string }
 */
apiRouter.post(
  "/api/stt",
  // Per-route raw body parser for binary audio (10MB limit matches WebSocket maxPayload)
  // Note: express.json() in index.ts does NOT interfere because Content-Type differs
  (req: Request, res: Response, next) => {
    // Collect raw binary body
    const chunks: Buffer[] = [];
    req.on("data", (chunk: Buffer) => chunks.push(chunk));
    req.on("end", () => {
      (req as Request & { rawBody: Buffer }).rawBody = Buffer.concat(chunks);
      next();
    });
    req.on("error", next);
  },
  async (req: Request, res: Response): Promise<void> => {
    try {
      const rawBody = (req as Request & { rawBody: Buffer }).rawBody;
      if (!rawBody || rawBody.length === 0) {
        res.status(400).json({ error: "Empty audio body" });
        return;
      }

      const audioData = await bufferToFloat32(rawBody);
      const sttPipe = getSttPipeline();
      const sttResult = await sttPipe(audioData, { chunk_length_s: 10 });
      const text = Array.isArray(sttResult)
        ? sttResult[0]?.text || ""
        : (sttResult as { text: string }).text;

      res.json({ text });
    } catch (err) {
      console.error("[/api/stt]", err);
      res.status(500).json({ error: String(err) });
    }
  },
);

/**
 * POST /api/llm
 * Body: { prompt: string }
 * Content-Type: application/json
 * Response: { response: string }
 */
apiRouter.post("/api/llm", async (req: Request, res: Response): Promise<void> => {
  try {
    const { prompt } = req.body as { prompt: string };
    if (!prompt || typeof prompt !== "string") {
      res.status(400).json({ error: "Missing or invalid 'prompt' field" });
      return;
    }

    const { text } = await generateText({ model: gemma3n, prompt });
    res.json({ response: text });
  } catch (err) {
    console.error("[/api/llm]", err);
    res.status(500).json({ error: String(err) });
  }
});

/**
 * POST /api/tts
 * Body: { text: string }
 * Content-Type: application/json
 * Response: 16-bit WAV binary
 * Content-Type: audio/wav
 */
apiRouter.post("/api/tts", async (req: Request, res: Response): Promise<void> => {
  try {
    const { text } = req.body as { text: string };
    if (!text || typeof text !== "string") {
      res.status(400).json({ error: "Missing or invalid 'text' field" });
      return;
    }

    const ttsPipe = getTtsPipeline();
    const ttsOut = await ttsPipe(text, { speaker_embeddings: SPEAKER_EMBEDDINGS });
    const wav = generateWav(ttsOut as { sampling_rate: number; audio: Float32Array });
    const buffer = wav.toBuffer() as Buffer;

    res.set("Content-Type", "audio/wav");
    res.set("Content-Length", String(buffer.length));
    res.send(buffer);
  } catch (err) {
    console.error("[/api/tts]", err);
    res.status(500).json({ error: String(err) });
  }
});
```
